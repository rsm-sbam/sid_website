# Load and clean data
df <- penguins[, c("bill_length_mm", "flipper_length_mm")] |> 
  drop_na()
X <- as.matrix(df)
# Define custom K-means functions
euclidean <- function(a, b) sqrt(sum((a - b)^2))

initialize_centroids <- function(X, k) {
  set.seed(42)
  X[sample(1:nrow(X), k), ]
}

assign_clusters <- function(X, centroids) {
  apply(X, 1, function(x) {
    which.min(apply(centroids, 1, function(c) euclidean(x, c)))
  })
}

update_centroids <- function(X, labels, k) {
  do.call(rbind, lapply(1:k, function(i) {
    points <- X[labels == i, , drop = FALSE]
    if (nrow(points) > 0) colMeans(points)
    else X[sample(1:nrow(X), 1), ]
  }))
}

kmeans_custom <- function(X, k, max_iter = 10) {
  centroids <- initialize_centroids(X, k)
  history <- list(centroids)
  
  for (i in 1:max_iter) {
    labels <- assign_clusters(X, centroids)
    centroids <- update_centroids(X, labels, k)
    history[[i + 1]] <- centroids
  }
  
  list(centroids = centroids, labels = labels, history = history)
}
# Run custom K-means
k <- 3
result <- kmeans_custom(X, k)
history <- result$history
# Visualize clustering progress step-by-step
plot_list <- map2(history, 0:(length(history) - 1), function(centroids, step) {
  cent_df <- as.data.frame(centroids)
  colnames(cent_df) <- c("bill_length_mm", "flipper_length_mm")
  
  ggplot(df, aes(x = bill_length_mm, y = flipper_length_mm)) +
    geom_point(alpha = 0.4) +
    geom_point(data = cent_df, aes(x = bill_length_mm, y = flipper_length_mm),
               color = "red", size = 3, shape = 4) +
    ggtitle(paste("Step", step)) +
    theme_minimal()
})

wrap_plots(plot_list)

After running both the custom K-means implementation and R’s built-in kmeans() function on the Palmer Penguins dataset (using bill_length_mm and flipper_length_mm), we observe that:

Cluster Assignments: While the exact cluster labels may differ (due to label switching), the overall grouping of data points is similar. Both algorithms form clusters that align with natural groupings in the data. Centroid Convergence: The final centroid coordinates from both implementations are very close, indicating that the custom algorithm correctly minimizes the within-cluster sum of squares over iterations. Convergence Steps: The custom implementation iteratively updates centroids using Euclidean distance and shows how clusters shift and stabilize, giving clear visual insight into the algorithm’s behavior. The built-in version achieves the same result more efficiently but without step-by-step visibility. Reproducibility: Setting the same random seed ensures both implementations start with similar initial centroids, allowing for a more direct comparison. In summary, the custom K-means algorithm successfully replicates the behavior of the built-in version, validating both its correctness and usefulness as a learning tool to understand clustering mechanics.

# Calculate WCSS (within-cluster sum of squares) and Silhouette for K = 2 to 7
wcss <- c()
avg_silhouette <- c()

for (k in 2:7) {
  kmeans_model <- kmeans(X, centers = k, nstart = 10)
  wcss[k - 1] <- kmeans_model$tot.withinss
  
  sil <- silhouette(kmeans_model$cluster, dist(X))
  avg_silhouette[k - 1] <- mean(sil[, 3])
}

metrics_df <- tibble(
  k = 2:7,
  wcss = wcss,
  silhouette = avg_silhouette
)

  geom_line() +

  theme_minimal()

# Implement KNN by hand
predict_knn <- function(train_x, train_y, test_x, k = 5) {
  predict_y <- vector("numeric", nrow(test_x))
  
  for (i in 1:nrow(test_x)) {
    dists <- sqrt(rowSums((t(t(train_x) - test_x[i, ]))^2))
    neighbors <- order(dists)[1:k]
    neighbor_classes <- train_y[neighbors]
    predict_y[i] <- names(sort(table(neighbor_classes), decreasing = TRUE))[1]
  }
  
  factor(predict_y, levels = levels(train_y))
}
